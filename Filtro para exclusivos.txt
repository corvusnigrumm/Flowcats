def is_subscriber_only(article_tag) - bool
    
    Detecta si un artículo es exclusivo para suscriptores.
    El badge 'ET' puede aparecer de varias formas en el HTML.
    
    article_html = str(article_tag)

    # ── Estrategia 1 clases CSS típicas del badge de suscriptor ──
    SUBSCRIBER_CLASSES = [
        badge-premium,
        badge-subscriber,
        badge-et,
        premium-badge,
        icon-et,
        label-et,
        suscriptor,
        paywall,
    ]
    for css_class in SUBSCRIBER_CLASSES
        if article_tag.find(class_=lambda c c and css_class in c.lower())
            return True

    # ── Estrategia 2 atributos data- de paywall ──
    PAYWALL_ATTRS = [data-premium, data-paywall, data-subscriber, data-et]
    for attr in PAYWALL_ATTRS
        if article_tag.find(attrs={attr True})
            return True

    # ── Estrategia 3 texto ET visible en un badge pequeño ──
    for tag in article_tag.find_all([span, div, i, em, label])
        text = tag.get_text(strip=True)
        if text == ET
            return True

    # ── Estrategia 4 keywords en el HTML crudo ──
    PAYWALL_KEYWORDS = [
        paywall, premium-content, subscriber-only,
        solo-suscriptores, exclusive-content
    ]
    for kw in PAYWALL_KEYWORDS
        if kw in article_html.lower()
            return True

    return False
